import java.util.Stack;

class Solution {
    public boolean isValid(String s) {
        // 创建一个栈，用来存放左括号
        Stack<Character> stack = new Stack<>();

        // 遍历字符串中的每一个字符
        for (int i = 0; i < s.length(); i++) {
            // 取出当前位置的字符
            char ch = s.charAt(i);

            // 如果当前字符是左括号，就入栈
            if (ch == '(' || ch == '{' || ch == '[') {
                stack.push(ch);
            } else {
                // 如果当前字符是右括号，但栈为空
                // 说明没有左括号可以和它匹配，直接返回 false
                if (stack.isEmpty()) {
                    return false;
                }

                // 弹出栈顶的左括号，准备和当前右括号匹配
                char top = stack.pop();

                // 如果当前是 )，但栈顶不是 (，说明不匹配
                if (ch == ')' && top != '(') {
                    return false;
                }

                // 如果当前是 ]，但栈顶不是 [，说明不匹配
                if (ch == ']' && top != '[') {
                    return false;
                }

                // 如果当前是 }，但栈顶不是 {，说明不匹配
                if (ch == '}' && top != '{') {
                    return false;
                }
            }
        }

        // 遍历结束后，如果栈为空，说明全部括号都成功匹配
        // 如果栈不为空，说明还有左括号没匹配完
        return stack.isEmpty();
    }
}
